cmake_minimum_required(VERSION 3.22)
project(aseprite_android_core LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(LAF_LIST_DIR "${ROOT}/laf")
set(LAF_BINARY_DIR "${CMAKE_CURRENT_BINARY_DIR}/laf")
set(LAF_WITH_TESTS OFF)
set(LAF_BACKEND "none")
if(NOT EXISTS "${ROOT}/laf/base/CMakeLists.txt" OR NOT EXISTS "${ROOT}/third_party/pixman/pixman/pixman.h")
  message(FATAL_ERROR "Missing upstream submodules. Run: git submodule update --init --recursive")
endif()
list(APPEND CMAKE_MODULE_PATH "${ROOT}/laf/cmake")
set(FMT_TEST OFF CACHE BOOL "" FORCE)
set(FMT_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/third_party/fmt" fmt)
add_subdirectory("${ROOT}/third_party/cityhash" cityhash)
add_subdirectory("${ROOT}/third_party/pixman-cmake" pixman)
set(PIXMAN_LIBRARY pixman)
set(PIXMAN_INCLUDE_DIR "${ROOT}/third_party/pixman/pixman" "${CMAKE_CURRENT_BINARY_DIR}")
add_subdirectory("${ROOT}/laf/base" laf-base)
target_include_directories(laf-base PUBLIC "${ROOT}/src")
target_compile_definitions(laf-base PUBLIC HAVE_CONFIG_H)
if(ANDROID)
  # Android's pthread implementation is in libc; there is no libpthread.so.
  set_property(TARGET laf-base PROPERTY LINK_LIBRARIES dl)
  set_property(TARGET laf-base PROPERTY INTERFACE_LINK_LIBRARIES dl)
  get_target_property(BASE_SRC laf-base SOURCES)
  list(FILTER BASE_SRC EXCLUDE REGEX "uuid_unix\\.cpp$")
  set_property(TARGET laf-base PROPERTY SOURCES "${BASE_SRC}")
  target_sources(laf-base PRIVATE "${CMAKE_CURRENT_LIST_DIR}/uuid_android.cpp")
endif()
add_subdirectory("${ROOT}/laf/gfx" laf-gfx)
if(ANDROID)
  set(ZLIB_LIBRARIES z)
else()
  find_package(ZLIB REQUIRED)
endif()
add_subdirectory("${ROOT}/src/fixmath" fixmath)
add_subdirectory("${ROOT}/src/doc" doc)
add_subdirectory("${ROOT}/src/render" render)
add_subdirectory("${ROOT}/src/flic" flic)
add_subdirectory("${ROOT}/src/dio" dio)
# These libraries retain upstream code and format implementations, not a recreation.
add_library(android-editor-core STATIC editor.cpp)
target_link_libraries(android-editor-core PUBLIC doc-lib render-lib dio-lib fmt)
target_include_directories(android-editor-core PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
if(ANDROID)
  add_library(aseprite_android SHARED bridge.cpp)
  target_link_libraries(aseprite_android PRIVATE android-editor-core log)
  target_link_options(aseprite_android PRIVATE "-Wl,-z,max-page-size=16384")
else()
  enable_testing()
  add_executable(core-tests core_tests.cpp)
  target_link_libraries(core-tests PRIVATE android-editor-core)
  add_test(NAME android-core-roundtrip COMMAND core-tests)
endif()
