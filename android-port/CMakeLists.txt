cmake_minimum_required(VERSION 3.20)
project(aseprite_android_native_port_modules LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_library(aseprite-android-touch STATIC src/android/touch_router.cpp)
target_include_directories(aseprite-android-touch PUBLIC src/android)
add_library(aseprite-android-raster STATIC src/android/raster_presenter.cpp)
target_include_directories(aseprite-android-raster PUBLIC src/android)
if(ANDROID)
  target_sources(aseprite-android-raster PRIVATE src/android/android_raster_window.cpp)
  target_link_libraries(aseprite-android-raster PUBLIC android)
  target_sources(aseprite-android-touch PRIVATE src/android/native_motion.cpp)
  target_link_libraries(aseprite-android-touch PUBLIC android)
endif()

# This bridge references the authentic Aseprite LAF types. It does not
# replace the window/system backend that remains to be implemented.
option(ASEPRITE_LAF_BRIDGE "Compile bridge against the original laf sources" OFF)
if(ASEPRITE_LAF_BRIDGE)
  if(NOT EXISTS "${ASEPRITE_SOURCE_DIR}/laf/os/event.h")
    message(FATAL_ERROR "Set ASEPRITE_SOURCE_DIR to a recursive Aseprite checkout")
  endif()
  add_library(aseprite-android-laf-bridge OBJECT src/android/laf_event_bridge.cpp)
  target_include_directories(aseprite-android-laf-bridge PRIVATE src/android "${ASEPRITE_SOURCE_DIR}/laf")
  target_link_libraries(aseprite-android-laf-bridge PRIVATE aseprite-android-touch)
endif()

if(NOT ANDROID)
  enable_testing()
  find_package(Threads REQUIRED)
  add_executable(touch-router-test tests/touch_router_test.cpp)
  target_link_libraries(touch-router-test PRIVATE aseprite-android-touch)
  add_test(NAME touch_router COMMAND touch-router-test)
  add_executable(event-queue-test tests/event_queue_test.cpp)
  target_include_directories(event-queue-test PRIVATE src/android)
  target_link_libraries(event-queue-test PRIVATE Threads::Threads)
  add_test(NAME event_queue COMMAND event-queue-test)
  add_executable(raster-presenter-test tests/raster_presenter_test.cpp)
  target_link_libraries(raster-presenter-test PRIVATE aseprite-android-raster)
  add_test(NAME raster_presenter COMMAND raster-presenter-test)
endif()
